# Focused follow-up: moderate oversold, recovery confirmation, and lower-churn long lookbacks.
# Formal five-year tests only; higher values are intended for the long side.
# The RSI/Bias/Drawdown band candidates target moderate rather than maximal oversold states.
OSR2-MODERATE-RSI35 ~ RANK(IF(RSI(CLOSE,14)<50,-ABS(RSI(CLOSE,14)-35),-99999)) ~ 1
OSR2-MODERATE-BIAS5 ~ RANK(-ABS(MIN(BIAS(CLOSE,20),0)+5)) ~ 1
OSR2-MODERATE-DD60 ~ RANK(-ABS((1 - CLOSE / TS_MAX(CLOSE,60)) - 0.10)) ~ 1
OSR2-RSI35-REC3 ~ (RANK(IF(RSI(CLOSE,14)<50,-ABS(RSI(CLOSE,14)-35),-99999)) + RANK(RETURNS(CLOSE,3))) / 2 ~ 1
OSR2-RSI35-REC5 ~ (RANK(IF(RSI(CLOSE,14)<50,-ABS(RSI(CLOSE,14)-35),-99999)) + RANK(RETURNS(CLOSE,5))) / 2 ~ 1
OSR2-DD60-REC5 ~ (RANK(1 - CLOSE / TS_MAX(CLOSE,60)) + RANK(RETURNS(CLOSE,5))) / 2 ~ 1
OSR2-MA20-REC5 ~ (RANK(1 - CLOSE / MA(CLOSE,20)) + RANK(CLOSE / MA(CLOSE,5) - 1)) / 2 ~ 1
OSR2-RSI-CROSS30 ~ RANK(1 - RSI(CLOSE,14) / 100) * AS_FLOAT(CROSS(RSI(CLOSE,14),30)) ~ 1
OSR2-PRICE-CROSS-MA5 ~ RANK(1 - CLOSE / MA(CLOSE,20)) * AS_FLOAT(CROSS(CLOSE,MA(CLOSE,5))) ~ 1
OSR2-RET20-REC5 ~ (RANK(1 - RETURNS(CLOSE,20)) + RANK(RETURNS(CLOSE,5))) / 2 ~ 1
OSR2-RET20 ~ RANK(1 - RETURNS(CLOSE,20)) ~ 1
OSR2-RET40 ~ RANK(1 - RETURNS(CLOSE,40)) ~ 1
OSR2-DD60 ~ RANK(1 - CLOSE / TS_MAX(CLOSE,60)) ~ 1
OSR2-DD120 ~ RANK(1 - CLOSE / TS_MAX(CLOSE,120)) ~ 1
OSR2-SMOOTH-RSI28 ~ RANK(MA(1 - RSI(CLOSE,28) / 100,10)) ~ 1
