def initialize(context):
    g.trading_signal = True
    g.pass_months = True
    g.run_stoploss = True
    g.hold_list = []
    g.yesterday_HL_list = []
    g.target_list = []
    g.not_buy_again = []
    g.history_hold_list = []
    g.stock_num = 5
    g.up_price = 80
    g.reason_to_sell = ''
    g.stoploss_limit = 0.07
    g.stoploss_market = 0.05
    g.limit_days = 5
    g.etf = '511880.XSHG'
    run_daily(prepare_stock_list, '9:00')
    run_weekly(weekly_adjustment, 2, '10:00')
    run_daily(sell_stocks, time='10:30')
    run_daily(sell_stocks, time='14:00')
    run_daily(trade_afternoon, time='14:30')
    run_daily(close_account, '14:30')

def prepare_stock_list(context):
    g.hold_list = list(context.portfolio.positions)
    g.not_buy_again = []
    if len(g.history_hold_list) >= g.limit_days:
        g.history_hold_list = g.history_hold_list[-g.limit_days:]
    if g.hold_list:
        df = get_price(g.hold_list, end_date=context.previous_date, frequency='daily', fields=['close', 'high_limit', 'low_limit'], count=1, panel=False, fill_paused=False)
        df = df[df['close'] == df['high_limit']]
        g.yesterday_HL_list = list(df.code)
    else:
        g.yesterday_HL_list = []
    g.trading_signal = today_is_pass(context)

def get_stock_list(context):
    final_list = []
    by_date = context.previous_date - datetime.timedelta(days=375)
    initial_list = get_index_stocks('399101.XSHE', by_date)
    initial_list = filter_kcbj_stock(initial_list)
    initial_list = filter_st_stock(initial_list)
    initial_list = filter_paused_stock(initial_list)
    initial_list = filter_limitup_stock(context, initial_list)
    initial_list = filter_limitdown_stock(context, initial_list)
    initial_list = filter_highprice_stock(context, initial_list)
    q = query(valuation.code, valuation.market_cap).filter(valuation.code.in_(initial_list), valuation.market_cap.between(5, 50)).order_by(valuation.market_cap.asc())
    df_fun = get_fundamentals(q)[:50]
    final_list = list(df_fun.code)
    return final_list

def weekly_adjustment(context):
    if g.trading_signal == True:
        g.target_list = get_stock_list(context)
        target_list = g.target_list[:g.stock_num]
        for stock in g.hold_list:
            if stock not in target_list and stock not in g.yesterday_HL_list:
                position = context.portfolio.positions[stock]
                if not close_position(position):
                    pass
        buy_security(context, target_list)

def sell_stocks(context):
    if g.run_stoploss:
        past_prices = history(2, '1d', 'close', get_index_stocks('399101.XSHE'))
        down_ratio = (past_prices.iloc[-1] / past_prices.iloc[0]).mean()
        if down_ratio <= 1 - g.stoploss_market:
            for stock in context.portfolio.positions:
                order_target_value_(stock, 0)
        else:
            for stock in context.portfolio.positions:
                if context.portfolio.positions[stock].price < context.portfolio.positions[stock].avg_cost * (1 - g.stoploss_limit):
                    order_target_value_(stock, 0)
                    g.reason_to_sell = 'stoploss'
                    g.not_buy_again.append(stock)

def check_limit_up(context):
    if g.yesterday_HL_list:
        current_data = get_current_data()
        for stock in g.yesterday_HL_list:
            if current_data[stock].last_price < current_data[stock].high_limit:
                position = context.portfolio.positions[stock]
                if close_position(position):
                    g.reason_to_sell = 'limitup'
                    g.not_buy_again.append(stock)
    g.history_hold_list.append(g.not_buy_again)

def check_remain_amount(context):
    if g.reason_to_sell in ['limitup', 'stoploss']:
        g.hold_list = list(context.portfolio.positions)
        if len(g.hold_list) < g.stock_num:
            target_list = g.target_list
            target_list = filter_not_buy_again(target_list)
            target_list = target_list[:min(g.stock_num, len(target_list))]
            buy_security(context, target_list)
        g.reason_to_sell = ''

def trade_afternoon(context):
    if g.trading_signal == True:
        check_limit_up(context)
        check_remain_amount(context)
    else:
        value = context.portfolio.available_cash
        if value > 0 and g.etf not in context.portfolio.positions:
            order_value(g.etf, value)

    temp_set = set()
    for hold_list in g.history_hold_list:
        temp_set = temp_set.union(set(hold_list))
    not_buy_again_list = list(temp_set)
    return [stock for stock in stock_list if stock not in not_buy_again_list]

def buy_security(context, target_list):
    position_count = len(context.portfolio.positions)
    target_num = len(target_list)
    if target_num > position_count:
        value = context.portfolio.available_cash / (target_num - position_count)
        for stock in target_list:
            if stock not in context.portfolio.positions:
                if open_position(stock, value):
                    if len(context.portfolio.positions) == target_num:
                        break

def today_is_pass(context):
    if g.pass_months:
        today = context.current_dt.strftime('%m-%d')
        if '04-05' <= today and today <= '04-30' or ('01-01' <= today and today <= '01-31'):
            return False
        else:
            return True
    else:
        return True

def close_account(context):
    if g.trading_signal == False:
        for stock in g.hold_list:
            if stock != g.etf:
                position = context.portfolio.positions[stock]
                if not close_position(position):
                    pass
