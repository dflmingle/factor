def before_market_open(context):
    if g.emptyholdday > 0:
        g.emptyholdday = g.emptyholdday - 1
    yesterday = context.previous_date
    list_date = get_before_after_trade_days(yesterday, g.lag1)
    g.ETFList = {}
    all_funds = get_all_securities(types='fund', date=yesterday)
    for idx in g.ETF_targets:
        symbol = g.ETF_targets[idx]
        if symbol in all_funds.index:
            if all_funds.loc[symbol].start_date <= list_date:
                g.ETFList[idx] = symbol
    return

def ETFtradeSell(context):
    if g.signal == 'CLEAR':
        for stock in context.portfolio.positions:
            if stock == g.empty_keep_stock:
                continue
            order_target(stock, 0)
    elif g.signal == 'BUY':
        if g.empty_keep_stock != '' and g.empty_keep_stock in context.portfolio.positions:
            order_target(g.empty_keep_stock, 0)
        holdings = set(context.portfolio.positions.keys())
        targets = set(g.buy)
        sells = holdings - targets
        for code in sells:
            order_target(code, 0)

def ETFtradeSell_back(context):
    if g.signal == 'CLEAR':
        for stock in context.portfolio.positions:
            if stock == g.empty_keep_stock:
                continue
            order_target(stock, 0)

def ETFtradeBuy(context):
    if g.emptyholdday > 0 and g.emptyholdday <= g.emptymaxday:
        return
    if g.signal == 'BUY':
        if g.empty_keep_stock != '' and g.empty_keep_stock in context.portfolio.positions:
            order_target(g.empty_keep_stock, 0)
        holdings = set(context.portfolio.positions.keys())
        targets = set(g.buy)
        ratio = len(targets)
        cash = context.portfolio.total_value / ratio
        adjusts = holdings & targets
        for code in adjusts:
            if abs(cash - context.portfolio.positions[code].value) > 5000:
                order_target_value(code, cash)
        purchases = targets - holdings
        for code in purchases:
            order_target_value(code, cash)
    if len(context.portfolio.positions) == 0 and g.empty_keep_stock and (g.empty_keep_stock != ''):
        order_target_value(g.empty_keep_stock, context.portfolio.available_cash)

def get_signal(context):
    buyold = g.buy
    if len(g.ETFList) == 0:
        before_market_open(context)
    df_etf = pd.DataFrame(columns=['基金代码', '对应指数', '周期涨幅', '均线差值', '最新涨幅', '今日涨幅'])
    current_data = get_current_data()
    for mkt_idx in g.ETFList:
        security = g.ETFList[mkt_idx]
        close_data = attribute_history(security, g.lag1, '1d', ['close'], df=False)
        current_price = current_data[security].last_price
        cp_increase = (current_price / close_data['close'][g.lag2 - g.lag1] - 1) * 100
        cp_increase_2 = (current_price / close_data['close'][-2] - 1) * 100
        curr_increase = (current_price / close_data['close'][-1] - 1) * 100
        ma_n1 = close_data['close'].mean()
        pre_price = (current_price / ma_n1 - 1) * 100
        df_etf = df_etf.append({'基金代码': security, '对应指数': mkt_idx, '周期涨幅': cp_increase, '均线差值': pre_price, '最新涨幅': cp_increase_2, '今日涨幅': curr_increase}, ignore_index=True)
    if len(df_etf) == 0:
        g.signal = 'CLEAR'
        return
    if g.ispython3:
        df_etf.sort_values(by='周期涨幅', ascending=False, inplace=True)
    else:
        df_etf.sort('周期涨幅', ascending=False, inplace=True)
    useidx = -1
    for i in range(0, len(df_etf['对应指数'])):
        if df_etf['周期涨幅'].iloc[i] >= 1 and df_etf['均线差值'].iloc[i] >= 0.1 and (df_etf['最新涨幅'].iloc[i] < g.emptyforthreeday) and (df_etf['最新涨幅'].iloc[i] > -g.emptyforthreeday * 0.5) and (df_etf['今日涨幅'].iloc[i] > -g.emptyforthreeday * 0.2) and (df_etf['今日涨幅'].iloc[i] < g.emptyforthreeday * 0.6) and (df_etf['周期涨幅'].iloc[i] < g.emptyforallday) and (df_etf['周期涨幅'].iloc[i] >= g.emptyforallday * 0.1):
            useidx = i
            break
    if useidx < 0:
        g.signal = 'CLEAR'
        return
    if g.use_dynamic_target_market:
        g.target_market = df_etf['对应指数'].iloc[useidx]
    red_etf_count = 0
    green_etf_count = 0
    for idx in df_etf['最新涨幅'].index:
        if df_etf['最新涨幅'].iloc[idx] > 0:
            red_etf_count = red_etf_count + 1
        else:
            green_etf_count = green_etf_count + 1
    if EmotionMonitor() == -1 or EmotionMonitorNewZZ500() == -1:
        g.signal = 'CLEAR'
        return
    if sum(df_etf['周期涨幅']) < 0.1 and green_etf_count > red_etf_count * 1.34:
        g.signal = 'CLEAR'
        g.emptyholdday = 1
        return
    if df_etf['周期涨幅'].iloc[useidx] < 0.1 or df_etf['均线差值'].iloc[useidx] < 0:
        g.signal = 'CLEAR'
        return
    g.buy = [df_etf['基金代码'].iloc[useidx]]
    for j in range(0, len(buyold)):
        for i in range(0, len(df_etf['对应指数'])):
            if df_etf['基金代码'].iloc[i] == buyold[j] and i < 5 and (useidx != i) and (df_etf['今日涨幅'].iloc[i] > g.emptyforthreeday * 0.01) and (df_etf['最新涨幅'].iloc[i] > g.emptyforthreeday * 0.1):
                g.buy = g.buy + [buyold[j]]
                break
    g.signal = 'BUY'
    return

def get_signal_back(context):
    current_data = get_current_data()
    for stock in context.portfolio.positions.keys():
        currs = context.portfolio.positions[stock]
        currper = (currs.price - currs.avg_cost) / currs.avg_cost * 100
        if currs.closeable_amount > 0:
            close_data = attribute_history(stock, 3, '1d', ['close'], df=False)
            current_price = current_data[stock].last_price
            curr_increase = (current_price / close_data['close'][-1] - 1) * 100
            two_increase = (current_price / close_data['close'][-2] - 1) * 100
            zz500ret = EmotionMonitorNewZZ500()
            if zz500ret == -1:
                order_target_value(stock, 0)
            elif currper > g.emptyforthreeday or two_increase > g.emptyforthreeday * 0.6 or curr_increase > g.emptyforthreeday * 0.4:
                order_target_value(stock, 0)
                g.emptyholdday = g.emptymaxday
            elif currper < -g.emptyforthreeday * 0.6 or two_increase < -g.emptyforthreeday * 0.4 or curr_increase < -g.emptyforthreeday * 0.2:
                order_target_value(stock, 0)
    return

def EmotionMonitor():
    _cnt = g.lag0 + max(g.lag, 3)
    volume = attribute_history(security=g.target_market, count=_cnt, unit='1d', fields='volume')['volume']
    v_ma_lag0 = talib.MA(np.array(volume), g.lag0)
    g.emotion_rate = round((volume[-1] / v_ma_lag0[-1] - 1) * 100, 2)
    vol_diff = volume - v_ma_lag0
    if vol_diff[-1] >= 0:
        ret_val = 1 if (vol_diff[-3:] >= 0).all() else 0
    else:
        ret_val = -1 if (vol_diff[-g.lag:] < 0).all() else 0
    return ret_val
