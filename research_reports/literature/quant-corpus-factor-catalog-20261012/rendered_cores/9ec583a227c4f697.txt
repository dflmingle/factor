            if sell_2 and (not nosell_1):
                _close_position(stock)
            else:
                current_percent = context.portfolio.positions[stock].value / context.portfolio.total_value
                if current_percent > maxpercent:
                    if order_target_value(stock, buycash):

def get_timing_signal(context, stock):
    g.mean_diff_day = 5
    close_data = attribute_history(g.ref_stock, g.mean_day + g.mean_diff_day, '1d', ['close'])
    high_low_data = attribute_history(g.ref_stock, g.N, '1d', ['high', 'low'])
    intercept, slope, r2 = get_ols(high_low_data.low, high_low_data.high)
    g.slope_series.append(slope)
    rsrs_score = get_zscore(g.slope_series[-g.M:]) * r2
    if g.singl == [] or g.singl == 'KEEP':
        if rsrs_score > g.score_threshold:
            g.singl = 'BUY'
        elif rsrs_score < -g.score_threshold:
            g.singl = 'SELL'
        else:
            g.singl = 'KEEP'
    elif g.singl == 'SELL':
        if rsrs_score <= -g.score_threshold * (1 - g.threshold):
            g.singl = 'SELL'
        else:
            g.singl = 'KEEP'
    elif g.singl == 'BUY':
        if rsrs_score >= g.score_threshold * (1 - g.threshold):
            g.singl = 'BUY'
        else:
            g.singl = 'KEEP'
    return g.singl
