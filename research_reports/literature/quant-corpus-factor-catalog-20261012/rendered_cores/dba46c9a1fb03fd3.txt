def get_stock_list(context):
    target_list, _ = prepare_stock_list(context)
    qualified_stocks = []
    sbgk_stocks = []
    current_data = get_current_data()
    date_now = context.current_dt.strftime('%Y-%m-%d')
    start = date_now + g.auction_start_time
    end = date_now + g.auction_end_time
    count_condition1 = 0
    count_condition2 = 0
    count_condition3 = 0
    for s in target_list:
        prev_day_data = attribute_history(s, 1, '1d', fields=['close', 'volume', 'money'], skip_paused=True)
        avg_price_increase_value = prev_day_data['money'][0] / prev_day_data['volume'][0] / prev_day_data['close'][0] * 1.1 - 1
        if avg_price_increase_value < g.avg_price_upper_limit or prev_day_data['money'][0] < g.sbgk_min_money or prev_day_data['money'][0] > g.sbgk_max_money:
            continue
        turnover_ratio_data = get_valuation(s, start_date=context.previous_date, end_date=context.previous_date, fields=['turnover_ratio', 'market_cap', 'circulating_market_cap'])
        if turnover_ratio_data.empty or turnover_ratio_data['market_cap'][0] < g.min_market_cap or turnover_ratio_data['circulating_market_cap'][0] > g.max_market_cap:
            continue
        count_condition1 += 1
        auction_data = get_call_auction(s, start_date=start, end_date=end, fields=['time', 'volume', 'current'])
        if auction_data.empty or auction_data['volume'][0] / prev_day_data['volume'][-1] < g.auction_vol_ratio:
            continue
        current_ratio = auction_data['current'][0] / prev_day_data['close'][-1]
        if current_ratio <= g.min_high_open_ratio or current_ratio >= g.max_high_open_ratio:
            continue
        count_condition2 += 1
        hst = attribute_history(s, g.left_pressure_history_days, '1d', fields=['high', 'volume'], skip_paused=True)
        prev_high = hst['high'].iloc[-1]
        zyts_0 = next((i - 1 for i, high in enumerate(hst['high'][-3::-1], 2) if high >= prev_high), 100)
        zyts = zyts_0 + g.left_pressure_buffer
        volume_data = hst['volume'][-zyts:]
        if len(volume_data) < 2 or volume_data[-1] <= max(volume_data[:-1]) * g.volume_break_ratio:
            continue
        count_condition3 += 1
        sbgk_stocks.append(s)
    qualified_stocks = sbgk_stocks
    return qualified_stocks
