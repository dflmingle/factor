def check_stocks(context):
    if g.check_stocks_days % g.check_stocks_refresh_rate != 0:
        g.check_stocks_days += 1
        return
    g.check_out_lists = get_security_universe(context, g.security_universe_index, g.security_universe_user_securities)
    g.check_out_lists = industry_filter(context, g.check_out_lists, g.industry_list)
    g.check_out_lists = concept_filter(context, g.check_out_lists, g.concept_list)
    g.check_out_lists = st_filter(context, g.check_out_lists)
    g.check_out_lists = delisted_filter(context, g.check_out_lists)
    g.check_out_lists = financial_statements_filter(context, g.check_out_lists)
    g.check_out_lists = situation_filter(context, g.check_out_lists)
    g.check_out_lists = technical_indicators_filter(context, g.check_out_lists)
    g.check_out_lists = pattern_recognition_filter(context, g.check_out_lists)
    g.check_out_lists = other_func_filter(context, g.check_out_lists)
    input_dict = get_check_stocks_sort_input_dict()
    g.check_out_lists = check_stocks_sort(context, g.check_out_lists, input_dict, g.check_out_lists_ascending)
    if context.current_dt.month in [2, 5, 8, 11]:
        g.check_out_lists = fund50(context, g.check_out_lists)
    g.check_stocks_days = 1
    return

def trade(context):
    if context.current_dt.month in [2, 5, 8, 11]:
        buy_lists = []
        if g.buy_trade_days % g.buy_refresh_rate == 0:
            buy_lists = g.check_out_lists
            buy_lists = st_filter(context, buy_lists)
            buy_lists = paused_filter(context, buy_lists)
            buy_lists = delisted_filter(context, buy_lists)
            buy_lists = high_limit_filter(context, buy_lists)
        if g.sell_trade_days % g.sell_refresh_rate != 0:
            g.sell_trade_days += 1
        else:
            sell(context, buy_lists)
            g.sell_trade_days = 1
        if g.buy_trade_days % g.buy_refresh_rate != 0:
            g.buy_trade_days += 1
        else:
            buy(context, buy_lists)
            g.buy_trade_days = 1

def strstock_numstock(stock):
    return str(stock[:6])

def fund50(context, stock_list):
    today = context.previous_date
    yesmonth = today + timedelta(days=-30)
    before_2month_start = today + timedelta(days=-120)
    before_2month_end = today + timedelta(days=-90)
    temp_list = []
    for stock in stock_list:
        stock_t = strstock_numstock(stock)
        df1 = finance.run_query(query(finance.FUND_PORTFOLIO_STOCK).filter(finance.FUND_PORTFOLIO_STOCK.symbol == stock_t, finance.FUND_PORTFOLIO_STOCK.pub_date < today, finance.FUND_PORTFOLIO_STOCK.pub_date > yesmonth).order_by(finance.FUND_PORTFOLIO_STOCK.pub_date.desc()))
        t1 = df1['market_cap'].sum()
        df2 = finance.run_query(query(finance.FUND_PORTFOLIO_STOCK).filter(finance.FUND_PORTFOLIO_STOCK.symbol == stock_t, finance.FUND_PORTFOLIO_STOCK.pub_date < before_2month_end, finance.FUND_PORTFOLIO_STOCK.pub_date > before_2month_start).order_by(finance.FUND_PORTFOLIO_STOCK.pub_date.desc()))
        t2 = df2['market_cap'].sum()
        temp_list.append([stock, t2, t1])
    fund_df = pd.DataFrame(temp_list, columns=['code', 'quarter2', 'quarter1'])
    fund_df['quarter2'] = fund_df['quarter2'].astype('int')
    fund_df['quarter1'] = fund_df['quarter1'].astype('int')
    fund_df = fund_df.sort_values(by=['quarter1'], ascending=False)
    return list(fund_df['code'])[:5]
