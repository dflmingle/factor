from jqdata import *

def initialize(context):
    g.buy_stock_count = 1
    g.check_out_lists = []
    g.etf = '511010.XSHG'
    g.yesterday_HL_list = []
    run_daily(before_market_open, time='9:00')
    run_monthly(check_stocks, 1, time='9:05')
    run_monthly(my_trade, 1, time='9:30')
    run_daily(check_limit_up, time='14:00')
    run_daily(check_limit_up, time='14:50')

def before_market_open(context):
    hold_list = [stock for stock in context.portfolio.positions if stock != g.etf]
    g.yesterday_HL_list.clear()
    if len(hold_list) != 0:
        df = get_price(hold_list, end_date=context.previous_date, frequency='daily', fields=['close', 'high_limit'], count=1, panel=False, fill_paused=False)
        df = df[df['close'] == df['high_limit']]
        g.yesterday_HL_list = list(df.code)

def my_trade(context):
    stocks = filter_limitup_stock(context, g.check_out_lists)
    stocks = filter_limitdown_stock(context, stocks)
    current_data = get_current_data()
    for stock in context.portfolio.positions:
        if stock in stocks:
            continue
        if stock in g.yesterday_HL_list:
            continue
        if current_data[stock].last_price < current_data[stock].high_limit:
            order_target(stock, 0)
        else:
            g.yesterday_HL_list.append(stock)
    position_count = len(context.portfolio.positions)
    if g.buy_stock_count > position_count:
        value = context.portfolio.cash / (g.buy_stock_count - position_count)
        for stock in stocks:
            if stock in context.portfolio.positions:
                continue
            if order_target_value(stock, value):
                if len(context.portfolio.positions) == g.buy_stock_count:
                    break

def check_limit_up(context):
    current_data = get_current_data()
    for stock in g.yesterday_HL_list:
        if current_data[stock].last_price < current_data[stock].high_limit:
            order_target(stock, 0)
    order_value(g.etf, context.portfolio.available_cash)

def check_stocks(context):
    current_data = get_current_data()
    check_date = context.previous_date - datetime.timedelta(days=200)
    all_stocks = list(get_all_securities(date=check_date).index)
    all_stocks = [stock for stock in all_stocks if not (current_data[stock].paused or current_data[stock].is_st or '退' in current_data[stock].name or stock.startswith('68') or stock.startswith('8') or stock.startswith('4'))]
    q = query(valuation.code, valuation.market_cap, valuation.pe_ratio, income.total_operating_revenue).filter(valuation.pb_ratio < 1, cash_flow.subtotal_operate_cash_inflow > 1000000.0, indicator.adjusted_profit > 1000000.0, indicator.roa > 0.15, indicator.inc_net_profit_year_on_year > 0, valuation.code.in_(all_stocks)).order_by(indicator.roa.desc()).limit(g.buy_stock_count * 3)
    check_out_lists = list(get_fundamentals(q).code)
    g.check_out_lists = check_out_lists[:g.buy_stock_count]

def filter_limitup_stock(context, stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list if stock in context.portfolio.positions or current_data[stock].last_price < current_data[stock].high_limit]

def filter_limitdown_stock(context, stock_list):
    current_data = get_current_data()
    return [stock for stock in stock_list if stock in context.portfolio.positions or current_data[stock].last_price > current_data[stock].low_limit]
