def filter_by_trend_strength(etf_pool, top_percent=0.6):
    scores = {}
    for etf in etf_pool:
        raw = trend_strength(etf)
        if raw:
            scores[etf] = raw
    if not scores:
        return []
    df = pd.DataFrame(scores).T
    rank_df = df.rank(pct=True) * 100
    total = rank_df['mom'] + rank_df['bias']
    threshold = np.percentile(total, (1 - top_percent) * 100)
    selected = total[total >= threshold].index.tolist()
    return selected

def trend_strength(security):
    df = attribute_history(security, 80, '1d', ['close'], skip_paused=True)
    if df is None or len(df) < 60:
        return None
    close = df['close'].values
    current = close[-1]
    mom_20 = current / close[-21] - 1 if len(close) > 20 else 0.0
    ma60 = np.mean(close[-60:])
    bias_60 = current / ma60 - 1
    return {'mom': mom_20, 'bias': bias_60}

def min_corr(stocks):
    nday = 245
    p = history(nday, '1d', 'close', stocks).dropna(axis=1)
    if len(p.columns) < 2:
        return list(stocks)[:4]
    r = np.log(p).diff()[1:]
    m_corr = r.corr()
    corr_mean = {}
    for stock in m_corr.columns:
        corr_abs = m_corr[stock].drop(stock).abs()
        if len(corr_abs) > 0:
            corr_mean[stock] = float(np.nanmean(corr_abs.values))
        else:
            corr_mean[stock] = 1.0
    sorted_stocks = sorted(corr_mean.keys(), key=lambda x: corr_mean[x])
    return sorted_stocks[:4]
