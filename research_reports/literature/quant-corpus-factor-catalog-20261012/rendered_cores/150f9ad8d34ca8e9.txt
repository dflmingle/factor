def epo(x, signal, lambda_, method='simple', w=None, anchor=None, normalize=True, endogenous=True):
    if isinstance(x, pd.Series):
        x = x.to_frame()
    n = x.shape[1]
    vcov = x.cov()
    corr = x.corr()
    I = np.eye(n)
    V = np.diag(np.diag(vcov))
    std = np.sqrt(V)
    s = signal
    a = anchor
    shrunk_cor = (1 - w) * I @ corr.values + w * I
    cov_tilde = std @ shrunk_cor @ std
    inv_shrunk_cov = solve(cov_tilde, I)
    if method == 'simple':
        epo = 1 / lambda_ * inv_shrunk_cov @ s
    elif method == 'anchored':
        if endogenous:
            gamma = np.sqrt(a.T @ cov_tilde @ a) / np.sqrt(s.T @ inv_shrunk_cov @ cov_tilde @ inv_shrunk_cov @ s)
            epo = inv_shrunk_cov @ ((1 - w) * gamma * s + w * I @ V @ a)
        else:
            epo = inv_shrunk_cov @ ((1 - w) * (1 / lambda_) * s + w * I @ V @ a)
    else:
        raise ValueError('`method` not accepted. Try `simple` or `anchored` instead.')
    if normalize:
        epo = [0 if a_val < 0 else a_val for a_val in epo]
        epo = epo / np.sum(epo)
    return epo

def run_optimization(stocks, end_date):
    if len(stocks) == 0:
        return {}
    if len(stocks) == 1:
        return {stocks[0]: 1.0}
    prices = get_price(stocks, count=250, end_date=end_date, frequency='daily', fields=['close'], panel=False)['close']
    if isinstance(prices, pd.Series):
        prices = prices.to_frame()
    returns = prices.pct_change().dropna()
    if isinstance(returns, pd.Series):
        returns = returns.to_frame()
    d = returns.var()
    if isinstance(d, pd.Series):
        inv_d = 1.0 / d
        a = inv_d / inv_d.sum()
    else:
        a = np.ones(len(stocks)) / len(stocks)
    total_return = prices.iloc[-1] / prices.iloc[0] - 1
    ranks = total_return.rank()
    momentum_weights = ranks / ranks.sum()
    composite = 0.5 * a + 0.5 * momentum_weights
    composite = composite / composite.sum()
    w = epo(x=returns, signal=composite, lambda_=10, method='simple', w=0.6)
    weights = dict(zip(stocks, w))
    return weights

def filter_new_stock(context, stock_list, d):
    yesterday = context.previous_date
    filtered = []
    removed = []
    for stock in stock_list:
        try:
            info = get_security_info(stock)
            start_date = info.start_date
            if yesterday - start_date < datetime.timedelta(days=d):
                removed.append(stock)
            else:
                filtered.append(stock)
        except Exception as e:
            removed.append(stock)
    for stk in filtered:

    return filtered
