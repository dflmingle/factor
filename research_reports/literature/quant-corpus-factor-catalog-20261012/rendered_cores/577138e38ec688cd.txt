def initial_slope_series():
    data = attribute_history(g.ref_stock, g.N + g.M, '1d', ['high', 'low'])
    multe_data = [get_ols(data.low[i:i + g.N], data.high[i:i + g.N]) for i in range(g.M)]
    slopes = [i[1] for i in multe_data]
    return slopes

def get_rsrs_score_mean(rsrs_score):
    N = 12
    g.rsrs_score_hisitory.append(rsrs_score)
    if len(g.rsrs_score_hisitory) < N:
        return 0
    return mean(g.rsrs_score_hisitory[-N:])

def get_timing_signal(context, stock):
    data = attribute_history(g.ref_stock, g.N, '1d', ['high', 'low', 'volume'])
    intercept, slope, r2 = get_ols(data.low, data.high)
    g.slope_series.append(slope)
    rsrs_score = get_zscore(g.slope_series[-g.M:]) * r2
    rsrs_mean = get_rsrs_score_mean(rsrs_score)
    if rsrs_score > 0 and rsrs_score < rsrs_mean:
        return 'SELL'
    if rsrs_score > -g.score_threshold:
        return 'BUY'
    else:
        return 'SELL'
