def get_stock_list(context):
    initial_list = get_index_stocks('000905.XSHG', date=None)
    initial_list = filter_new_stock(context, initial_list)
    initial_list = filter_kcb_stock(context, initial_list)
    initial_list = filter_st_stock(initial_list)
    q = query(valuation.code, valuation.circulating_market_cap).filter(valuation.code.in_(initial_list)).order_by(valuation.circulating_market_cap.asc())
    df = get_fundamentals(q)
    shizhi_list = list(df.code)
    num_1 = int(len(shizhi_list) / 4)
    shizhi_list1 = shizhi_list[num_1:-1]
    indus = get_industries(name='sw_l1', date=context.current_dt)
    hy_code_list = indus.index.tolist()
    peg_all = []
    for hy_code in hy_code_list:
        i_stocklist1 = get_industry_stocks(hy_code, date=None)
        hyshizhi_list = [x for x in i_stocklist1 if x in shizhi_list1]
        peg_list = []
        if hyshizhi_list:
            peg_list = get_factor_filter_list(context, hyshizhi_list, 'PEG', True, 0, 0.7)
        for code in peg_list:
            peg_all.append(code)
    npgr_all = []
    for hy_code in hy_code_list:
        i_stocklist1 = get_industry_stocks(hy_code, date=None)
        hyshizhi_list = [x for x in i_stocklist1 if x in shizhi_list1]
        npgr_list = []
        if hyshizhi_list:
            npgr_list = get_factor_filter_list(context, hyshizhi_list, 'net_profit_growth_rate', False, 0, 0.7)
        for code in npgr_list:
            npgr_all.append(code)
    roe_all = []
    for hy_code in hy_code_list:
        i_stocklist1 = get_industry_stocks(hy_code, date=None)
        hyshizhi_list = [x for x in i_stocklist1 if x in shizhi_list1]
        q1 = query(indicator.code, indicator.roe).filter(indicator.code.in_(hyshizhi_list)).order_by(indicator.roe.desc())
        df1 = get_fundamentals(q1)
        roe_list = list(df1.code)
        num_2 = int(len(roe_list) / 3)
        roe_list1 = roe_list[0:num_2]
        for code in roe_list1:
            roe_all.append(code)
    PNR_all = []
    for i in peg_all:
        if i in npgr_all and i in roe_all:
            PNR_all.append(i)
    q3 = query(valuation.code, valuation.circulating_market_cap).filter(valuation.code.in_(PNR_all)).order_by(valuation.circulating_market_cap.desc())
    df3 = get_fundamentals(q3)
    result_list = list(df3.code)
    return result_list
