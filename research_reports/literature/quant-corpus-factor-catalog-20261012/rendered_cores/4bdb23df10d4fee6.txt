def remove_new_stocks(security_list, context):
    for stock in security_list:
        days_public = (context.current_dt.date() - get_security_info(stock).start_date).days
        if days_public < 100:
            security_list.remove(stock)
    return security_list

def filter_stock_ST(stock_list):
    curr_data = get_current_data()
    for stock in stock_list:
        if curr_data[stock].is_st or 'ST' in curr_data[stock].name or '*' in curr_data[stock].name or ('退' in curr_data[stock].name):
            stock_list.remove(stock)
    return stock_list
