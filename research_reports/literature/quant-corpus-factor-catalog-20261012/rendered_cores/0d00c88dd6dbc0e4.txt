def transform_date(date, format_type='d'):
    if isinstance(date, str):
        if format_type == 'd':
            return datetime.datetime.strptime(date, '%Y-%m-%d').date()
    elif isinstance(date, datetime):
        if format_type == 'd':
            return date.date()
    return date

def filter_new_stock(initial_list, date, days=90):
    d_date = transform_date(date, 'd')
    result = []
    for stock in initial_list:
        try:
            security_info = get_security_info(stock)
            if security_info and d_date - security_info.start_date > dt.timedelta(days=days):
                result.append(stock)
        except:
            continue
    return result

def filter_st_paused_stock(initial_list, date):
    current_data = get_current_data()

                continue
            if not stock_data.is_st and (not stock_data.paused) and ('退' not in str(stock_data.name)):
                result.append(stock)

            continue
    return result

def filter_paused_stock(initial_list, date):
    if not initial_list:
        return []
    try:
        df = get_price(initial_list, end_date=date, frequency='daily', fields=['paused'], count=1, panel=False, fill_paused=True)
        df = df[df['paused'] == 0]
        paused_list = list(df.code)
        return paused_list
    except:
        return initial_list

def get_hl_stock(initial_list, date):
    if not initial_list:
        return []
    try:
        df = get_price(initial_list, end_date=date, frequency='daily', fields=['close', 'high_limit'], count=1, panel=False, fill_paused=False, skip_paused=False)
        df = df.dropna()
        df = df[abs(df['close'] - df['high_limit']) / df['high_limit'] < 0.002]
        hl_list = list(df.code)
        return hl_list
    except:
        return []

def get_trading_days_since(context, start_date_str):
    try:
        start_date = datetime.datetime.strptime(start_date_str, '%Y-%m-%d').date()
        current_date = context.current_dt.date()
        if start_date >= current_date:
            return 0
        trade_days = get_trade_days(start_date=start_date, end_date=current_date)
        return len(trade_days) - 1
    except Exception as e:
        return 999

def is_reversal_candle(df):
    if df is None or len(df) < 2:
        return False
    latest = df.iloc[-1]
    prev = df.iloc[-2]
    candle_body = abs(latest['close'] - latest['open'])
    candle_body_ratio = candle_body / latest['open'] if latest['open'] != 0 else 0
    if latest['close'] > latest['open']:
        lower_shadow = latest['open'] - latest['low']
        upper_shadow = latest['high'] - latest['close']
    else:
        lower_shadow = latest['close'] - latest['low']
        upper_shadow = latest['high'] - latest['open']
    prev_pct = (prev['close'] - df.iloc[-3]['close']) / df.iloc[-3]['close'] if len(df) >= 3 else 0
    is_doji = candle_body_ratio < 0.01 and lower_shadow > 0 and (upper_shadow > 0)
    is_small_yang = latest['close'] > latest['open'] and candle_body_ratio < 0.03
    is_hammer = lower_shadow > candle_body * 2 and upper_shadow < candle_body * 0.3 and (prev_pct < -0.02)
    is_morning_star = prev['close'] < prev['open'] and (prev['open'] - prev['close']) / prev['open'] > 0.02 and (latest['close'] > latest['open']) and (candle_body_ratio < 0.02)
    return is_doji or is_small_yang or is_hammer or is_morning_star

def before_market_open(context):
    current_date = context.current_dt.strftime('%Y-%m-%d')
    yesterday = context.previous_date.strftime('%Y-%m-%d')
    stock_list = get_all_securities(['stock']).index.tolist()
    stock_list = filter_kcbj_stock(stock_list)
    stock_list = filter_new_stock(stock_list, yesterday, days=90)
    stock_list = filter_st_paused_stock(stock_list, current_date)
    hl_stocks = get_hl_stock(stock_list, yesterday)
    board_count = 0
    daily_boards = []
    for stock in hl_stocks:
        df = attribute_history(stock, 120, '1d', ['close', 'high', 'low', 'volume', 'money'], skip_paused=True)
        if len(df) < 60:
            continue
        yesterday_data = df.iloc[-1]
        prev_day_data = df.iloc[-2]
        if yesterday_data['close'] > g.params['max_price'] or yesterday_data['close'] < g.params['min_price']:
            continue
        pct_change = (yesterday_data['close'] / prev_day_data['close'] - 1) * 100
        turnover = yesterday_data['volume'] * yesterday_data['close'] / yesterday_data['money'] if yesterday_data['money'] > 0 else 0
        ma60 = df['close'].mean()
        vol_ma20 = df['volume'][-20:].mean()
        limit_days = 0
        for i in range(1, 26):
            if i >= len(df):
                break
            day_data = df.iloc[-1 - i]
            prev_day = df.iloc[-2 - i] if i + 1 < len(df) else None
            if prev_day is not None:
                day_high_limit = get_high_limit(stock, prev_day['close'])
                if abs(day_data['close'] - day_high_limit) / day_high_limit < 0.001:
                    limit_days += 1
        is_first_limit = limit_days == 0
        is_not_one_word = turnover > 0.03
        is_low_position = yesterday_data['close'] > ma60 * 0.9 and yesterday_data['close'] < ma60 * 1.3
        is_volume_spike = yesterday_data['volume'] > vol_ma20 * 2
        if is_not_one_word and is_low_position and is_volume_spike and is_first_limit:
            g.first_board[stock] = {'date': current_date, 'board_date': yesterday, 'volume': yesterday_data['volume'], 'price': yesterday_data['close'], 'ma60': ma60, 'pct_change': pct_change, 'turnover': turnover}
            board_count += 1
            daily_boards.append(stock)

def get_high_limit(stock, prev_close):
    if stock.startswith('30'):
        return round(prev_close * 1.2, 2)
    elif stock.startswith('68'):
        return round(prev_close * 1.2, 2)
    elif stock.startswith('4') or stock.startswith('8'):
        return round(prev_close * 1.3, 2)
    else:
        return round(prev_close * 1.1, 2)

def check_buy_signals(context):
    current_date = context.current_dt.strftime('%Y-%m-%d')
    buy_list = []
    for stock, info in list(g.first_board.items()):
        if stock in context.portfolio.positions or stock in g.buy_signals:
            continue
        days_passed = get_trading_days_since(context, info['board_date'])
        if days_passed < 2:
            continue
        if days_passed > g.params['max_wait_days']:
            del g.first_board[stock]
            continue
        current_data = get_current_data()
        if stock not in current_data or current_data[stock].paused:
            continue
        current_price = current_data[stock].last_price
        board_price = info['price']
        drop_percent = (board_price - current_price) / board_price * 100
        if drop_percent < 6 or drop_percent > 15:
            continue
        df = attribute_history(stock, 30, '1d', ['close', 'high', 'low', 'volume', 'open'], skip_paused=True)
        if len(df) < 20:
            continue
        latest = df.iloc[-1]
        prev = df.iloc[-2]
        ma5 = df['close'][-5:].mean()
        ma10 = df['close'][-10:].mean()
        ma20 = df['close'][-20:].mean()
        vol_shrink = latest['volume'] < info['volume'] * 0.6
        near_ma10 = abs(latest['close'] - ma10) / ma10 < 0.02
        near_ma20 = abs(latest['close'] - ma20) / ma20 < 0.02
        is_below_ma20 = latest['close'] < ma20 * (1 - 0.02)
        near_support = is_below_ma20
        reversal = is_reversal_candle(df)
        is_below_ratio = 0.02
        is_below_ma20 = latest['close'] < ma20 * (1 - is_below_ratio)
        price_up = prev is not None and latest['close'] > prev['close']
        if len(df) >= 4:
            recent_lows = df['low'].iloc[-4:-1].min()
            low_not_new = latest['low'] > recent_lows * 0.99
        else:
            low_not_new = True
        if len(df) >= 5:
            lows_3days = df['low'].iloc[-5:-2].min()
            lows_2days = df['low'].iloc[-3:-1].min()
            tow_days_new_low = lows_2days > lows_3days * 0.99
        else:
            tow_days_new_low = True
        con1 = [price_up, low_not_new, tow_days_new_low]
        score1 = sum(con1)
        reversal = is_reversal_candle(df)
        if vol_shrink and is_below_ma20 and (reversal or score1 >= 2):
            g.buy_signals[stock] = {'date': current_date, 'price': latest['close'], 'stop_loss_ratio': g.params['stop_loss_ratio'], 'ma5': ma5, 'ma10': ma10, 'ma20': ma20, 'days_passed': days_passed, 'support_line': f'跌破20日线{0.02 * 100:.0f}%'}
            buy_list.append(stock)
            pct_from_board = (latest['close'] / info['price'] - 1) * 100
    if buy_list:
        buy_list.sort(key=lambda x: abs(g.buy_signals[x]['days_passed'] - 4))
        num = g.params['num'] - len(context.portfolio.positions)
        buy_list = buy_list[:num]
        execute_buy(context, buy_list)

def execute_buy(context, buy_list):
    current_date = context.current_dt.strftime('%Y-%m-%d')
    cash = context.portfolio.available_cash
    if cash <= 0 or len(buy_list) == 0:
        return
    per_stock_cash = cash / g.params['num']
    for stock in buy_list:
        if stock not in g.buy_signals:
            continue
        signal_info = g.buy_signals[stock]
        if signal_info['date'] != current_date:
            continue
        current_data = get_current_data()
        if stock not in current_data:
            continue
        current_price = current_data[stock].last_price
        amount = int(per_stock_cash / (current_price * 100)) * 100
        if amount >= 100:
            order_target_value(stock, per_stock_cash)
            stop_loss_price = current_price * signal_info['stop_loss_ratio']
            g.positions[stock] = {'buy_date': current_date, 'buy_price': current_price, 'stop_loss': stop_loss_price, 'stop_loss_ratio': signal_info['stop_loss_ratio'], 'ma5': signal_info['ma5'], 'ma10': signal_info['ma10']}
            del g.buy_signals[stock]
            if stock in g.first_board:
                del g.first_board[stock]

def check_sell_signals(context):
    if len(context.portfolio.positions) == 0:
        return
    current_date = context.current_dt.strftime('%Y-%m-%d')
    for stock in list(context.portfolio.positions.keys()):
        position = context.portfolio.positions[stock]
        if position.total_amount == 0:
            continue
        current_data = get_current_data()
        if stock not in current_data:
            continue
        current_price = current_data[stock].last_price
        df = attribute_history(stock, 10, '1d', ['close', 'high', 'low', 'volume'], skip_paused=True)
        if len(df) < 5:
            continue
        ma5 = df['close'][-5:].mean()
        sell_signal = False
        reason = ''
        if stock in g.positions:
            if current_price >= position.avg_cost * (1 + g.params['profit_target_ratio']):
                sell_signal = True
                profit_pct = (current_price / position.avg_cost - 1) * 100
                reason = f'止盈触发 (盈利{profit_pct:.1f}%)'
        if stock in g.positions:
            stop_loss = g.positions[stock]['stop_loss']
            if current_price < stop_loss:
                sell_signal = True
                loss_pct = (current_price / position.avg_cost - 1) * 100
                reason = f'止损触发 (亏损{loss_pct:.1f}%)'
        if not sell_signal:
            latest = df.iloc[-1]
            prev = df.iloc[-2]
            avg_vol = df['volume'][-5:-1].mean() if len(df) >= 5 else df['volume'].mean()
            vol_surge = latest['volume'] > avg_vol * 1.5
            price_drop = (latest['close'] - prev['close']) / prev['close'] < -0.04
            if vol_surge and price_drop:
                sell_signal = True
                reason = '高位放量下跌'
        if sell_signal:
            profit_pct = (current_price / position.avg_cost - 1) * 100
            if not hasattr(g, 'sold_stocks'):
                g.sold_stocks = {}
            buy_date = g.positions.get(stock, {}).get('buy_date', 'unknown')
            g.sold_stocks[stock] = {'stock': stock, 'sell_date': current_date, 'sell_price': current_price, 'sell_reason': reason, 'profit_pct': profit_pct / 100, 'buy_date': buy_date, 'buy_price': position.avg_cost}
            order_target(stock, 0)
            if stock in g.positions:
                del g.positions[stock]
