def my_Trader(context):
    dt_last = context.previous_date
    stocks = get_all_securities('stock', dt_last).index.tolist()
    stocks = filter_kcbj_stock(stocks)
    stocks = filter_st_stock(stocks)
    df = get_fundamentals(query(valuation.code).filter(valuation.code.in_(stocks), valuation.pb_ratio > 0, indicator.inc_return > 0, indicator.inc_total_revenue_year_on_year > 0, indicator.inc_net_profit_year_on_year > 0, indicator.ocf_to_operating_profit > 5).order_by(valuation.market_cap.asc()).limit(g.stock_num))
    choice = list(df.code)
    for s in context.portfolio.positions:
        if s not in choice and s not in g.high_limit_list:
            order_target(s, 0)
    psize = 1.0 / g.stock_num * context.portfolio.total_value
    for s in choice:
        if context.portfolio.available_cash < psize:
            break
        if s not in context.portfolio.positions:
            order_value(s, psize)

def prepare_stock_list(context):
    g.high_limit_list = []
    hold_list = list(context.portfolio.positions)
    if hold_list:
        df = get_price(hold_list, end_date=context.previous_date, frequency='daily', fields=['close', 'high_limit', 'paused'], count=1, panel=False)
        g.high_limit_list = df.query('close==high_limit and paused==0')['code'].tolist()

def check_limit_up(context):
    current_data = get_current_data()
    if g.high_limit_list:
        for stock in g.high_limit_list:
            if current_data[stock].last_price < current_data[stock].high_limit:
                order_target(stock, 0)
