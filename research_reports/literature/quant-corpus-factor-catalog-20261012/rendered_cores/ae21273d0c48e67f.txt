def commodity_etf_trade(context, total_value):
    end_date = context.previous_date
    if end_date.year < 2020:
        g.commodity_etf = ['162411.XSHE', '518880.XSHG', '511010.XSHG']
    else:
        g.commodity_etf = ['162411.XSHE', '518880.XSHG', '159985.XSHE']
    weights = run_optimization(g.commodity_etf, end_date)
    if weights is None:
        return
    index = 0
    for w in weights:
        value = total_value * w
        order_target_value(g.commodity_etf[index], value)
        index += 1

def filter_DQII_etf(context):
    lofs = get_all_securities(['etf'])
    lofs = lofs[lofs['start_date'] < context.previous_date]
    lofs = lofs[lofs['end_date'] > context.current_dt.date()]
    lofs = lofs.index.tolist()
    current_data = get_current_data()
    lofs = [stock for stock in lofs if ('德' in current_data[stock].name or '纳' in current_data[stock].name or '法' in current_data[stock].name or ('DAX' in current_data[stock].name) or ('标' in current_data[stock].name) or ('英' in current_data[stock].name) or ('日' in current_data[stock].name) or ('印' in current_data[stock].name) or ('越' in current_data[stock].name) or ('225' in current_data[stock].name)) and '中' not in current_data[stock].name and ('企' not in current_data[stock].name)]
    net_value = get_extras('unit_net_value', lofs, end_date=context.previous_date, df=True, count=1).T
    net_value.columns = ['unit_net_value']
    close = history(count=1, unit='1d', field='close', security_list=lofs).T
    volume = history(count=1, unit='1d', field='volume', security_list=lofs).T
    volume = volume[volume > 1000000]
    close = close[close.index.isin(volume.index)]
    close.columns = ['close']
    val = net_value.join(close)
    val['score'] = val['close'] / val['unit_net_value']
    val = val[val['score'] > 0]
    val = val[val['score'] < 1.1]
    val = val.sort_values(by='score', ascending=False)
    buy_list = val.index.tolist()
    return buy_list

def portfolio_variance(weights, cov_matrix):
    return np.dot(weights.T, np.dot(cov_matrix * 250, weights))

def optimize_portfolio(returns):
    cov_matrix = returns.cov()
    num_assets = len(returns.columns)
    w_min = 1 / np.power(num_assets, 2)
    w_max = 0.9
    init_weights = np.array([1 / num_assets] * num_assets)
    weight_sum_constraint = {'type': 'eq', 'fun': lambda weights: np.sum(weights) - 1}
    bnds = tuple(((w_min, w_max) for x in range(num_assets)))
    result = minimize(portfolio_variance, init_weights, method='SLSQP', args=cov_matrix, bounds=bnds, constraints=weight_sum_constraint)
    return result.x

def run_optimization(stocks, end_date):
    prices = get_price(stocks, count=250, end_date=end_date, frequency='daily', fields=['close'])['close']
    returns = prices.pct_change().dropna()
    weights = optimize_portfolio(returns)
    return weights
