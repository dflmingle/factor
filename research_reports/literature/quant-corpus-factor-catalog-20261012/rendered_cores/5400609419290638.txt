    n_sh = finance.run_query(query(finance.STK_ML_QUOTA).filter(finance.STK_ML_QUOTA.day <= yesterday, finance.STK_ML_QUOTA.link_id == 310001).order_by(finance.STK_ML_QUOTA.day.desc()).limit(10))
    if n_sh.empty:
        return 0
    n_sz = finance.run_query(query(finance.STK_ML_QUOTA).filter(finance.STK_ML_QUOTA.day <= yesterday, finance.STK_ML_QUOTA.link_id == 310002).order_by(finance.STK_ML_QUOTA.day.desc()).limit(10))
    total_net_in = 0
    if n_sz.empty:
        for i in range(0, 1):
            sh_in = n_sh['buy_amount'][i] - n_sh['sell_amount'][i]
            total_net_in += sh_in
    else:
        for i in range(0, 1):
            sh_in = n_sh['buy_amount'][i] - n_sh['sell_amount'][i]

def getPreDayPrice(security):
    close_data = attribute_history(security, 1, '1d', ['close'])
    current_price = close_data['close'][-1]
    return current_price

def print_code_info(context, code, value):
    huadian = 0.015
    for position in list(context.portfolio.positions.values()):
        if str(position.security) == code:
            if value == 0:
                account = position.total_amount * -1
                price = round(position.price * (1 - huadian), 2)
                return order(code, account, LimitOrderStyle(price))
            elif value == position.value:
                pass
            elif value > position.value:
                price = round(position.price * (1 + huadian), 2)
                account = (value - position.total_amount) / price
                return order(code, account, LimitOrderStyle(price))
            elif value < position.value:
                price = round(position.price * (1 - huadian), 2)
                account = (value - position.total_amount) / price
                return order(code, account, LimitOrderStyle(price))
            return
    price = round(getPreDayPrice(code) * (1 + huadian), 2)
    account = value / price
    return order(code, account, LimitOrderStyle(price))

def order_target_value_(security, value, context):
    if value == 0:
        pass
    return print_code_info(context, security, value)

def open_position(security, value, context):
    order = order_target_value_(security, value, context)
    if order != None and order.filled > 0:
        return True
    return False

def close_position(position, context):
    security = position.security
    order = order_target_value_(security, 0, context)
    if order != None:
        if order.status == OrderStatus.held and order.filled == order.amount:
            return True
    return False
