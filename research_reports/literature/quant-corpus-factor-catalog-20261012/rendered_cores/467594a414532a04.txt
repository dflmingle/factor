class BiLSTM(nn.Module):

    def __init__(self):
        super(BiLSTM, self).__init__()
        self.n_class = 6
        self.n_hidden = 10
        self.lstm = nn.LSTM(input_size=self.n_class, hidden_size=self.n_hidden, bidirectional=True)
        self.fc = nn.Linear(self.n_hidden * 2, 1)

    def forward(self, X):
        batch_size = X.shape[0]
        input = X.transpose(0, 1)
        hidden_state = torch.randn(1 * 2, batch_size, self.n_hidden)
        cell_state = torch.randn(1 * 2, batch_size, self.n_hidden)
        outputs, (_, _) = self.lstm(input, (hidden_state, cell_state))
        outputs = outputs[-1]
        model = self.fc(outputs)
        return model
