def initialize(context):
    g.up_level = 2
    g.factors_config = [{'factor': 'market_cap', 'sort': True}, {'factor': 'VOL5', 'sort': False}]
    g.stop_loss = -7
    g.holds = 3
    run_daily(get_stock_list, '9:10')
    run_daily(buy, '09:30')
    run_daily(sell_v3, '14:50')
    run_daily(print_position_info, '15:02')

def get_stock_list(context):
    date = context.previous_date
    date = transform_date(date, 'str')
    initial_list = prepare_stock_list(date)
    hl_list = get_hl_stock(initial_list, date)
    ccd = get_continue_count_df(hl_list, date, 20) if len(hl_list) != 0 else pd.DataFrame(index=[], data={'count': [], 'extreme_count': []})
    if len(ccd) > 0:
        sorted_ccd = ccd.sort_values(by='count', ascending=False)
        rank = 1
        for stock in sorted_ccd.index:
            count = sorted_ccd.loc[stock, 'count']
            extreme = sorted_ccd.loc[stock, 'extreme_count']
            name = get_security_info(stock).display_name
            rank += 1
    max_count = ccd['count'].max()
    threshold = max(2, max_count - g.up_level)
    candidate_stocks = ccd[ccd['count'] >= threshold].index.tolist()
    df = get_factor_filter_df_V2(context, candidate_stocks, 'market_cap', 'VOL5', True, False, top_n=10)
    stock_list = list(df.index)
    g.target_list = stock_list[:g.holds - len(context.portfolio.positions)]

def buy(context):
    current_data = get_current_data()
    value = context.portfolio.total_value / g.holds
    for s in g.target_list:
        if context.portfolio.available_cash / current_data[s].last_price > 100:
            if current_data[s].last_price == current_data[s].high_limit:
                order_value(s, value, LimitOrderStyle(current_data[s].day_open))
                name = get_security_info(s).display_name
            else:
                order_value(s, value, MarketOrderStyle(current_data[s].day_open))
                name = get_security_info(s).display_name

def sell_v3(context):
    hold_list = list(context.portfolio.positions)
    current_data = get_current_data()
    for s in hold_list:
        position = context.portfolio.positions[s]
        name = get_security_info(s).display_name
        cost = position.avg_cost
        price = position.price
        ret = 100 * (price / cost - 1)
        current_value = position.value
        is_limit_up = current_data[s].last_price == current_data[s].high_limit
        is_limit_down = current_data[s].last_price == current_data[s].low_limit
        if is_limit_up:
            continue
        if is_limit_down:
            continue
        start_date = transform_date(position.init_time, 'str')
        current_date = transform_date(context.current_dt, 'str')
        start_dt = dt.datetime.strptime(start_date, '%Y-%m-%d')
        current_dt = dt.datetime.strptime(current_date, '%Y-%m-%d')
        hold_days = (current_dt - start_dt).days
        sell_reason = ''
        sell_amount = 0
        if ret < g.stop_loss:
            sell_amount = current_value
            sell_reason = f'止损({ret:.2f}%)'
        elif ret >= 20:
            sell_amount = current_value
            sell_reason = f'止盈({ret:.2f}%)'
        elif ret >= 10:
            sell_amount = current_value * 0.5
            sell_reason = f'部分止盈({ret:.2f}%)'
        elif ret >= 5:
            sell_amount = current_value * 0.3
            sell_reason = f'初步止盈({ret:.2f}%)'
        elif hold_days >= 5 and ret < 0:
            sell_amount = current_value
            sell_reason = f'时间止损({hold_days}天)'
        if sell_amount > 0:
            if sell_amount >= current_value * 0.95:
                order_target_value(s, 0)
            else:
                target_value = current_value - sell_amount
                order_target_value(s, target_value)

def sell(context):
    hold_list = list(context.portfolio.positions)
    current_data = get_current_data()
    for s in hold_list:
        name = get_security_info(s).display_name
        if not current_data[s].last_price == current_data[s].high_limit:
            if context.portfolio.positions[s].closeable_amount != 0:
                start_date = transform_date(context.portfolio.positions[s].init_time, 'str')
                target_date = get_shifted_date(start_date, 2, 'T')
                current_date = transform_date(context.current_dt, 'str')
                cost = context.portfolio.positions[s].avg_cost
                price = context.portfolio.positions[s].price
                ret = 100 * (price / cost - 1)
                sell_reason = []
                if current_date >= target_date:
                    sell_reason.append(f'持有时间达标(买入日:{start_date})')
                if ret > 0:
                    sell_reason.append(f'已盈利({ret:.2f}%)')
                if sell_reason:
                    if current_data[s].last_price > current_data[s].low_limit:
                        order_target_value(s, 0)

def get_factor_filter_df(context, stock_list, jqfactor, sort):
    if len(stock_list) != 0:
        yesterday = context.previous_date
        score_list = get_factor_values(stock_list, jqfactor, end_date=yesterday, count=1)
        score_list = score_list[jqfactor].iloc[0].tolist()
        df = pd.DataFrame(index=stock_list, data={'score': score_list}).dropna()
        df = df.sort_values(by='score', ascending=sort)
    else:
        df = pd.DataFrame(index=[], data={'score': []})
    return df

def get_factor_filter_df_V1(context, stock_list, factors_config):
    if len(stock_list) == 0:
        return pd.DataFrame()
    yesterday = context.previous_date
    current_stocks = stock_list.copy()
    df_factors = pd.DataFrame(index=current_stocks)
    for i, factor_info in enumerate(factors_config):
        factor_code = factor_info['factor']
        factor_name = f'factor_{i}'
        factor_values = get_factor_values(current_stocks, factor_code, end_date=yesterday, count=1)
        df_factors[factor_name] = factor_values[factor_code].iloc[0].tolist()
    df_factors = df_factors.dropna()
    current_df = df_factors.copy()
    for i, factor_info in enumerate(factors_config):
        factor_name = f'factor_{i}'
        sort_ascending = factor_info['sort']
        current_df = current_df.sort_values(by=factor_name, ascending=sort_ascending)
        factor_code = factor_info['factor']
    return current_df

def get_factor_filter_df_V2(context, stock_list, jqfactor1, jqfactor2, sort1, sort2, top_n=100):
    if len(stock_list) != 0:
        yesterday = context.previous_date
        factor1_values = get_factor_values(stock_list, jqfactor1, end_date=yesterday, count=1)
        factor2_values = get_factor_values(stock_list, jqfactor2, end_date=yesterday, count=1)
        df = pd.DataFrame(index=stock_list)
        df['factor1'] = factor1_values[jqfactor1].iloc[0].tolist()
        df['factor2'] = factor2_values[jqfactor2].iloc[0].tolist()
        df = df.dropna()
        df = df.sort_values(by='factor1', ascending=sort1)
        df_top = df.head(top_n)
        df_top = df_top.sort_values(by='factor2', ascending=sort2)
        return df_top
    else:
        return pd.DataFrame()
