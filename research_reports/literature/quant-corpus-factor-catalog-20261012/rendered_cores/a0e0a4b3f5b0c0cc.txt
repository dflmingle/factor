def getRankedFactors(f, d, previousdate):
    far = get_factor_values(g.all_stocks, factors=f, end_date=previousdate, count=1)
    first_factor_name = f[0]
    for i in f:
        if i == first_factor_name:
            df = far[i].T
            df.columns = [first_factor_name]
        else:
            df1 = far[i].T
            df1.columns = [i]
            df = df.join(df1)
    df['code'] = df.index
    res = [[0] * len(f) for i in range(len(df))]
    for i in range(0, len(df)):
        for j in range(0, len(f)):
            res[i][j] = df[f[j]][i]
    fillNan(res)
    getRank(res)
    return (res, df['code'])
