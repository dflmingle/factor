def filter_st_paused_new(stock_list, days, context):
    df = get_all_securities(types=['stock'], date=context.current_dt)
    kcb = list(df[df.index.str.startswith('688')].index.unique())
    start_date = (context.current_dt - timedelta(days=days)).date()
    df_new_stock = df[df['start_date'] > start_date]
    stock_list = list(set(stock_list).difference(set(df_new_stock.index)).difference(set(kcb)))
    curr_data = get_current_data()
    stock_list = [stock for stock in stock_list if not curr_data[stock].is_st]
    stock_list = [stock for stock in stock_list if not curr_data[stock].paused]
    stock_list = [curr_data[stock].code for stock in stock_list if '退' not in curr_data[stock].name]
    return stock_list
