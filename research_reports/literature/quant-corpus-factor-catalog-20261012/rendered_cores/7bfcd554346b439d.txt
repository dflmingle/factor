def filter_aiguo_stock(context, stock_list):
    current_data = get_current_data()
    keywords = ['爱', '我', '中', '华']
    LIST = []
    for stock in stock_list:
        for KEY in current_data[stock].name:
            if str(KEY) in keywords:
                LIST.append(stock)
    return LIST
