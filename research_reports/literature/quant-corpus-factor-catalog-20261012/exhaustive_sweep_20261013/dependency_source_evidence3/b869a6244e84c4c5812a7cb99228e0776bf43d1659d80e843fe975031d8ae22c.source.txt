from jqdata import *
import numpy as np

def initialize(context):
    set_benchmark('000300.XSHG')
    set_option('use_real_price', True)
    
    # 策略参数
    g.buy_days = 3  # 涨停后观察天数5,8
    g.callback_min = -0.12  # 回调幅度下限
    g.callback_max = -0.08  # 回调幅度上限
    g.stop_ratio = 2.0  # 止盈止损比例
    
    g.monitored = {}  # 监控的涨停股票
    g.buy_signals = []  # 买入信号
    
    run_daily(market_open, 'open')
    run_daily(check_signals, '14:50')
    run_daily(process_trades, '14:55')

def market_open(context):
    """开盘时检查止损止盈"""
    for stock in list(context.portfolio.positions.keys()):
        pos = context.portfolio.positions[stock]
        if pos.closeable_amount <= 0:
            continue
        
        # 获取当前价格
        current_data = get_current_data()
        if stock not in current_data:
            continue
            
        current_price = current_data[stock].last_price
        
        # 检查止损
        stop_info = g.monitored.get(stock, {})
        if 'stop_price' in stop_info and current_price <= stop_info['stop_price']:
            order_target(stock, 0)
            if stock in g.monitored:
                del g.monitored[stock]
            log.info(f"止损卖出 {stock}: 现价{current_price:.2f} <= 止损价{stop_info['stop_price']:.2f}")
            continue
            
        # 检查止盈（止损价的2倍）
        if 'stop_price' in stop_info and 'buy_price' in stop_info:
            target_price = stop_info['stop_price'] + 2 * (stop_info['buy_price'] - stop_info['stop_price'])
            if current_price >= target_price:
                order_target(stock, 0)
                if stock in g.monitored:
                    del g.monitored[stock]
                log.info(f"止盈卖出 {stock}: 现价{current_price:.2f} >= 目标价{target_price:.2f}")

def check_signals(context):
    """收盘前检查买入信号"""
    g.buy_signals = []
    
    # 检查监控的涨停股是否满足买入条件
    for stock, info in list(g.monitored.items()):
        if stock in context.portfolio.positions:
            continue
            
        # 超过监控天数则移除
        if (context.current_dt - info['date']).days > g.buy_days:
            del g.monitored[stock]
            continue
        
        # 检查回调条件
        if check_buy_condition(context, stock, info):
            g.buy_signals.append(stock)

def check_buy_condition(context, stock, info):
    """检查买入条件"""
    # 获取当日数据
    today_data = get_price(stock, end_date=context.current_dt, count=1, 
                          fields=['close', 'low', 'high', 'open'], skip_paused=True)
    
    if len(today_data) == 0:
        return False
    
    close_price = today_data['close'][0]
    open_price = today_data['open'][0]
    low_price = today_data['low'][0]
    
    # 计算回调幅度
    callback_ratio = (close_price - info['limit_close']) / info['limit_close']
    
    # 条件1: 回调在-8%到-12%之间
    cond1 = (callback_ratio >= g.callback_min and callback_ratio <= g.callback_max)
    
    # 条件2: 收盘价在10日均线上方
    ma10_data = get_price(stock, end_date=context.current_dt, count=10, 
                         fields=['close'], skip_paused=True)
    if len(ma10_data) < 10:
        return False
    
    ma10 = ma10_data['close'].mean()
    cond2 = close_price > ma10
    
    if cond1 and cond2:
        # 记录买入信息
        info['buy_date'] = context.current_dt
        info['buy_price'] = close_price  # 当日收盘价作为买入参考
        info['stop_price'] = low_price   # 当日最低价作为止损价
        
        log.info(f"买入信号 {stock}: 收盘价{close_price:.2f}, 回调幅度{callback_ratio:.2%}, "
                f"MA10={ma10:.2f}, 止损价{low_price:.2f}")
        return True
    
    return False

def process_trades(context):
    """处理交易"""
    # 寻找新的涨停股
    find_limit_up_stocks(context)
    
    # 执行买入
    for stock in g.buy_signals:
        if stock in context.portfolio.positions:
            continue
            
        if context.portfolio.cash > 0:
            # 获取当前价格（使用当日收盘价作为买入价）
            hist_data = get_price(stock, end_date=context.current_dt, count=1,
                                fields=['close'], skip_paused=True)
            if len(hist_data) == 0:
                continue
                
            buy_price = hist_data['close'][0]
            
            # 计算买入数量（使用可用资金的20%）
            buy_value = context.portfolio.cash * 0.2
            amount = int(buy_value // buy_price)
            
            # 确保买入数量是100的整数倍
            amount = max(100, amount // 100 * 100)
            
            if amount >= 100 and buy_price > 0:
                # 记录实际买入价
                g.monitored[stock]['actual_buy'] = buy_price
                order(stock, amount)
                log.info(f"买入 {stock}: {amount}股 @ {buy_price:.2f}, 总金额{buy_price*amount:.2f}")

def find_limit_up_stocks(context):
    """寻找涨停（首板，非一字）股票"""
    # 获取所有正常交易的股票
    stocks = get_all_securities(['stock']).index.tolist()
    
    for stock in stocks:
        # 跳过ST股票
        if is_st_stock(stock):
            continue
            
        # 检查是否已监控
        if stock in g.monitored:
            continue
        
        # 检查是否涨停
        if check_first_limit_up(context, stock):
            # 记录涨停信息
            limit_data = get_price(stock, end_date=context.current_dt, count=1,
                                 fields=['close', 'open', 'high', 'low'], skip_paused=True)
            
            if len(limit_data) > 0:
                g.monitored[stock] = {
                    'date': context.current_dt,
                    'limit_close': limit_data['close'][0],
                    'limit_open': limit_data['open'][0],
                    'limit_high': limit_data['high'][0],
                    'limit_low': limit_data['low'][0]
                }
                
                log.info(f"监控涨停股 {stock}: "
                        f"涨停价{limit_data['close'][0]:.2f}, "
                        f"开盘{limit_data['open'][0]:.2f}")

def check_first_limit_up(context, stock):
    """检查是否为首板非一字涨停"""
    # 获取当日和前一日数据
    hist = get_price(stock, end_date=context.current_dt, count=2,
                    fields=['close', 'open', 'high', 'low'], skip_paused=True)
    
    if len(hist) < 2:
        return False
    
    # 今日数据
    today_close = hist['close'][-1]
    today_open = hist['open'][-1]
    
    # 昨日数据
    yesterday_close = hist['close'][-2]
    
    # 计算涨停价
    limit_price = calculate_limit_price(stock, yesterday_close)
    
    # 条件1: 今日收盘价达到涨停价
    cond1 = abs(today_close - limit_price) / limit_price < 0.001
    
    # 条件2: 非一字涨停（开盘价不等于涨停价）
    cond2 = abs(today_open - limit_price) / limit_price > 0.001
    
    # 条件3: 检查近期是否有涨停（排除连板）
    recent_data = get_price(stock, end_date=context.previous_date, count=10,
                           fields=['close'], skip_paused=True)
    if len(recent_data) >= 10:
        prev_closes = recent_data['close'].values
        prev_limit_counts = 0
        for i in range(1, len(prev_closes)):
            prev_limit = calculate_limit_price(stock, prev_closes[i-1])
            if abs(prev_closes[i] - prev_limit) / prev_limit < 0.001:
                prev_limit_counts += 1
        cond3 = prev_limit_counts == 0  # 近期没有涨停
    else:
        cond3 = True
    
    return cond1 and cond2 and cond3

def calculate_limit_price(stock, prev_close):
    """计算涨停价"""
    if prev_close <= 0:
        return prev_close
    
    # 根据股票类型确定涨停幅度
    if stock.startswith(('300', '688')):  # 创业板/科创板
        limit_ratio = 1.2
    elif stock.startswith('8'):  # 北交所
        limit_ratio = 1.3
    else:  # 主板
        if is_st_stock(stock):
            limit_ratio = 1.05
        else:
            limit_ratio = 1.1
    
    # 计算并四舍五入到分
    limit_price = round(prev_close * limit_ratio, 2)
    
    # A股规则：价格分位数处理
    if limit_ratio == 1.1:  # 主板
        limit_price = round(limit_price * 100) / 100
    elif limit_ratio == 1.2:  # 创业板/科创板
        limit_price = round(limit_price * 100) / 100
    elif limit_ratio == 1.05:  # ST股
        limit_price = round(limit_price * 100) / 100
    
    return limit_price

def is_st_stock(stock):
    """判断是否为ST股票"""
    try:
        info = get_security_info(stock)
        if hasattr(info, 'display_name'):
            name = info.display_name
        else:
            name = info.name
        return 'ST' in name or '*' in name or '退' in name
    except:
        return False
