# Oversold-rebound hypotheses; five-year-only formal tests.
# Higher values are intended to mean deeper oversold / stronger rebound potential.
OSR-RET5-20D ~ RANK(1 - RETURNS(CLOSE,5)) ~ 1
OSR-RET10-20D ~ RANK(1 - RETURNS(CLOSE,10)) ~ 1
OSR-DD20-20D ~ RANK(1 - CLOSE / TS_MAX(CLOSE,20)) ~ 1
OSR-MA20-20D ~ RANK(1 - CLOSE / MA(CLOSE,20)) ~ 1
OSR-RSI14-20D ~ RANK(1 - RSI(CLOSE,14) / 100) ~ 1
OSR-MFI14-20D ~ RANK(1 - MFI(CLOSE,HIGH,LOW,VOLUME,14) / 100) ~ 1
OSR-Z20-20D ~ RANK(1 - TS_ZSCORE(CLOSE,20)) ~ 1
OSR-RET5-RSI14-20D ~ (RANK(1 - RETURNS(CLOSE,5)) + RANK(1 - RSI(CLOSE,14) / 100)) / 2 ~ 1
OSR-RET5-DD20-20D ~ (RANK(1 - RETURNS(CLOSE,5)) + RANK(1 - CLOSE / TS_MAX(CLOSE,20))) / 2 ~ 1
OSR-CAPITULATION-20D ~ RANK(1 - RETURNS(CLOSE,5)) + RANK(MA(TURNOVER,5) / MA(TURNOVER,20)) ~ 1
OSR-SCALED-RET5-20D ~ RANK((1 - RETURNS(CLOSE,5)) / (STDDEV(RETURNS(CLOSE,1),20) + 0.0001)) ~ 1
